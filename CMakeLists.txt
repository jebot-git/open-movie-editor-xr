cmake_minimum_required(VERSION 3.21)
project(OpenMovieEditor VERSION 0.0.20090105 LANGUAGES C CXX)
set(CMAKE_INSTALL_DOCDIR "share/doc/openmovieeditor" CACHE STRING "Documentation location")
include(GNUInstallDirs)
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
set(DE_XR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "de-xr collection")
set(ARC_FLTK "${DE_XR_ROOT}/arc-sbs/fltk")
set(OME_DEPENDENCY_PREFIX "${CMAKE_CURRENT_SOURCE_DIR}/.deps/install" CACHE PATH "Gmerlin dependency prefix")
list(PREPEND CMAKE_PREFIX_PATH "${OME_DEPENDENCY_PREFIX}" "${CMAKE_CURRENT_SOURCE_DIR}/.deps/sysroot/usr")
set(ENV{PKG_CONFIG_PATH} "${OME_DEPENDENCY_PREFIX}/lib/pkgconfig:${CMAKE_CURRENT_SOURCE_DIR}/.deps/sysroot/usr/lib64/pkgconfig:$ENV{PKG_CONFIG_PATH}")
pkg_check_modules(MEDIA REQUIRED IMPORTED_TARGET gavl gmerlin libquicktime portaudio-2.0 jack samplerate sndfile glib-2.0 libavformat libavcodec libavutil libswscale libswresample)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
file(GLOB_RECURSE OME_SOURCES CONFIGURE_DEPENDS src/*.cxx src/*.cpp src/*.c)
list(REMOVE_ITEM OME_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/nle_main.cxx")
# Compile the upstream UI, timeline, effects, project manager and render dialogs.
add_library(ome-engine STATIC ${OME_SOURCES})
target_compile_features(ome-engine PUBLIC cxx_std_17)
target_include_directories(ome-engine PUBLIC "${ARC_FLTK}" src src/tinyxml src/sl icons "${CMAKE_CURRENT_BINARY_DIR}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/config.h" "#define VERSION \"0.0.20090105-qsv-sbs\"\n#define PACKAGE \"openmovieeditor\"\n#define HAVE_STDINT_H 1\n#define HAVE_INTTYPES_H 1\n#define HAVE_UNISTD_H 1\n#define HAVE_STDLIB_H 1\n#define HAVE_STRING_H 1\n#define HAVE_MALLOC 1\n")
target_compile_definitions(ome-engine PUBLIC INSTALL_PREFIX="${CMAKE_INSTALL_PREFIX}" __STDC_CONSTANT_MACROS)
target_link_libraries(ome-engine PUBLIC fltk::fltk-shared fltk::gl-shared fltk::images-shared PkgConfig::MEDIA PkgConfig::FONTCONFIG m mvec dl pthread)
add_executable(openmovieeditor src/nle_main.cxx)
target_include_directories(openmovieeditor PRIVATE "${ARC_FLTK}")
target_link_libraries(openmovieeditor PRIVATE ome-engine)
include("${ARC_FLTK}/desktop-menu.cmake")
de_xr_desktop_menu(openmovieeditor)
get_filename_component(OME_FLTK_LIBDIR "${FLTK_DIR}/../../lib" REALPATH)
set_target_properties(openmovieeditor PROPERTIES INSTALL_RPATH "${OME_FLTK_LIBDIR};${CMAKE_INSTALL_FULL_LIBEXECDIR}/openmovieeditor/lib")
configure_file(packaging/openmovieeditor.in openmovieeditor-launcher @ONLY)
install(TARGETS openmovieeditor RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/openmovieeditor)
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/openmovieeditor-launcher" DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME openmovieeditor)
configure_file(packaging/org.openmovieeditor.Editor.desktop org.openmovieeditor.Editor.desktop @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/org.openmovieeditor.Editor.desktop" DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(DIRECTORY doc/ DESTINATION ${CMAKE_INSTALL_DOCDIR} FILES_MATCHING PATTERN "*.html" PATTERN "*.png")
install(FILES COPYING README.XR.md THIRD_PARTY.XR.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
enable_testing()

# Keep the original codec/effects features, with private user-local runtimes.
install(FILES "${OME_DEPENDENCY_PREFIX}/lib/libgmerlin.so.0.0.0" DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/openmovieeditor/lib RENAME libgmerlin.so.0)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/.deps/sysroot/usr/lib64/libquicktime.so.0.0.0" DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/openmovieeditor/lib RENAME libquicktime.so.0)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/.deps/sysroot/usr/lib64/libdv.so.4.0.3" DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/openmovieeditor/lib RENAME libdv.so.4)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/.deps/sysroot/usr/lib64/libschroedinger-1.0.so.0.11.0" DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/openmovieeditor/lib RENAME libschroedinger-1.0.so.0)
install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/.deps/sysroot/usr/lib64/libquicktime/" DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/openmovieeditor/lib/libquicktime FILES_MATCHING PATTERN "*.so")
install(DIRECTORY "${OME_DEPENDENCY_PREFIX}/lib/gmerlin/plugins/" DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/openmovieeditor/lib/gmerlin/plugins FILES_MATCHING PATTERN "*.so")
install(FILES packaging/openmovieeditor.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/mime/packages)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/.deps/gmerlin-1.2.0/COPYING" DESTINATION ${CMAKE_INSTALL_DOCDIR} RENAME COPYING.Gmerlin)

foreach(package libquicktime libdv schroedinger)
  install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/.deps/sysroot/usr/share/licenses/${package}" DESTINATION ${CMAKE_INSTALL_DOCDIR}/licenses)
endforeach()
install(FILES packaging/dependency-rpms.sha256 scripts/bootstrap-legacy DESTINATION ${CMAKE_INSTALL_DOCDIR})
